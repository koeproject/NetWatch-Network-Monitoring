    if value is not None and value.tzinfo is None:

5. ขั้น 1.2 — Connector schema

เป้าหมายแปลง NDJSON ที่ engine push มาเป็น object ที่ validate แล้ว โดยเขียนจาก payload จริง ไฟล์backend/app/zabbix/connector_schema.py, backend/tests/fixtures/connector/*.ndjson (copy จาก lab/ndjson-sink/captured/), backend/tests/test_connector_schema.py

โค้ดหลัก

class _Line(BaseModel):
    model_config = ConfigDict(extra="ignore", frozen=True)

class ValueLine(_Line):
    host: HostRef
    itemid: int
    name: str
    clock: int
    ns: int = 0
    value: int | float | str      # int ก่อน float: counter แบบ unsigned ไม่เสียความแม่น
    type: int                     # value_type: 0 float, 1 char, 2 log, 3 unsigned, 4 text

class ProblemLine(_Line):
    value: Literal[1]
    eventid: int; clock: int; ns: int = 0
    name: str; severity: int; hosts: list[HostRef]

class RecoveryLine(_Line):
    value: Literal[0]
    eventid: int; p_eventid: int; clock: int; ns: int = 0

EventLine = Annotated[ProblemLine | RecoveryLine, Field(discriminator="value")]
  • extra="ignore": patch release อาจเพิ่ม field ใหม่ เราไม่ให้ระบบพังเพราะเรื่องนี้
  • ทุก field มาจากไฟล์ใน captured/ (ดู docs/connector-payload.md) ไม่มีตัวไหนมาจาก "เอกสารบอกว่า"
  • value: int | float | str: ใน payload จริง ค่าของ item แบบ float มาเป็น JSON integer ได้ (11 จาก 23 บรรทัดเป็น 0 ไม่ใช่ 0.0) และ Pydantic โหมด smart จะเลือก type ที่ตรงที่สุด
  • discriminator="value": problem กับ recovery มีหน้าตาต่างกัน Pydantic ดูค่า value (1 หรือ 0) แล้วเลือก model ให้ถูกตัว ถ้าเป็นค่าอื่นจะ error แล้วบรรทัดนั้นถูกข้าม
def parse_ndjson(body: bytes, adapter: TypeAdapter[T], *, kind: str) -> list[T]:
    parsed: list[T] = []
    for number, raw in enumerate(body.splitlines(), start=1):
        if not raw.strip():
            continue
        try:
            parsed.append(adapter.validate_python(json.loads(raw)))
        except (ValueError, ValidationError) as exc:
            log.warning("connector %s line %d skipped: %s | %.200r", kind, number, ...)
    return parsed
  • parse ทีละบรรทัด ถ้าบรรทัดไหนเสียก็ log แล้วข้าม ไม่ทิ้งทั้ง batch
  • json.JSONDecodeError เป็นลูกของ ValueError จึงจับได้ใน except เดียว
  • %.200r ตัด log ไว้ที่ 200 ตัวอักษร กันบรรทัดยักษ์ท่วม log

def _status(host) -> DeviceStatus:
    if str(host.get("status")) == "1":              return DeviceStatus.DISABLED     # not monitored
    if str(host.get("maintenance_status")) == "1":  return DeviceStatus.MAINTENANCE
    states = {str(i.get("available")) for i in host.get("interfaces", [])}
    if "1" in states: return DeviceStatus.UP        # available
    if "2" in states: return DeviceStatus.DOWN      # unavailable
    return DeviceStatus.UNKNOWN

Decision ที่ปิดในขั้นนี้ (ADR 0008): login ด้วย username + password แล้วได้ session token ที่ลงลายเซ็น HMAC ในรูป cookie แบบ HttpOnly (หรือส่งเป็น Bearer สำหรับ script) ยังไม่มี SSO

passwords.py: scrypt จาก stdlib

def hash_password(password: str) -> str:
    salt = secrets.token_bytes(16)
    digest = hashlib.scrypt(password.encode(), salt=salt, n=2**14, r=8, p=1, dklen=32)
    return f"scrypt${n}${r}${p}${salt.hex()}${digest.hex()}"
  • scrypt ตั้งใจให้ ช้าและกินหน่วยความจำ การเดารหัสจำนวนมากจึงแพง ส่วน salt ทำให้รหัสเดียวกันได้ hash ต่างกัน
  • เก็บพารามิเตอร์ไว้ใน hash ทำให้วันหน้าเพิ่มความแรงได้โดย hash เก่ายังใช้ได้

payload   = base64url(JSON {"u": username, "t": tenant_id, "exp": unix_seconds})

token     = payload + "." + signature

def verify(self, token):

11. ขั้น 1.8 — Live updates: bus, SSE, cache

เป้าหมายเกิดปัญหาแล้วหน้าเว็บรู้ทันที แม้ request push กับ SSE จะอยู่คนละ uvicorn worker ไฟล์backend/app/bus/memory.py, redis_bus.py, backend/app/services/stream.py, backend/app/api/stream.py, backend/app/cache.py

ทำไมต้องมี Redis

             worker #1                         worker #2
  POST /hooks/events ──► ingest ──publish──►  Redis channel "acme:live" ──► subscribe ──► SSE ──► browser

bus แบบ memory ส่งได้แค่ใน worker เดียวกัน ถ้า push เข้า worker #1 แต่ browser ต่ออยู่กับ worker #2 ข้อความจะไม่ถึง Dockerfile จึงตั้ง --workers 2 โดยตั้งใจ เพื่อให้การทดสอบพิสูจน์เส้นทางผ่าน Redis จริง

bus/memory.py: queue มีขนาดจำกัด

async def publish(self, message):
    for queue in self._subscribers:
        try: queue.put_nowait(message)
        except asyncio.QueueFull: log.warning("bus subscriber is full; message dropped")

browser ที่ค้าง (ไม่อ่าน) จะทำข้อความหาย แทนที่จะทำให้ memory โตไม่จำกัด เมื่อ reconnect แล้ว portal ค่อยอ่านสถานะใหม่จาก REST

services/stream.py: กรองตาม tenant + heartbeat

async with self._bus.subscribe() as messages:
    while True:
        if pending is None:
            pending = asyncio.ensure_future(anext(messages))
        done, _ = await asyncio.wait({pending}, timeout=self._heartbeat)
        if not done:
            yield HEARTBEAT; continue
        message, pending = pending.result(), None
        if message.get("device_ref") in refs:
            yield message
  • bus มีข้อความของทุก tenant ตรงนี้คือจุดที่แต่ละ connection ได้เฉพาะของตัวเอง
  • ถ้าไม่มีข้อความ 15 วินาทีจะส่ง heartbeat เพื่อกัน proxy ตัด connection ที่เงียบ และให้ browser รู้เมื่อ connection ตาย
  • ใช้ asyncio.wait กับ task ที่ค้างไว้ แทน wait_for เพราะ wait_for จะ cancel การอ่านทุกครั้งที่ timeout และอาจทำข้อความหาย
  • รายชื่ออุปกรณ์ถูก refresh ทุก 60 วินาที อุปกรณ์ใหม่จึงเริ่ม stream ได้โดยไม่ต้อง reconnect

14. Decision ที่ปิดใน Phase นี้

ADRเรื่องตัดสินใจผลที่ต้องรู้
0006ข้างใน device_refชื่อ host เชิงเทคนิค (host.host)payload ไม่มี hostid · เปลี่ยนชื่อ host = อุปกรณ์ใหม่ ให้ถือชื่อ host เป็น ID ถาวร
0007store อยู่ที่ไหน + เก็บนานเท่าไรDB acme + role acme_app ใน PG instance เดียวกัน, REVOKE CONNECT, ข้อมูลดิบเก็บ 30 วันต้องยืนยัน retention กับลูกค้าก่อน Phase 4 · ย้ายไป container แยกได้ด้วย pg_dump
